// Command server runs the calculator REST API.
//
// Configuration comes from the environment: PORT (default 8080). With the
// -healthcheck flag, it instead probes /healthz on the local server and exits
// 0 or 1, which lets the distroless image run a Docker healthcheck without a
// shell or curl.
package main

import (
        "context"
        "errors"
        "flag"
        "fmt"
        "io"
        "log/slog"
        "net"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "github.com/fabianguidobaldi/sezzle-challenge/backend/internal/api"
        "github.com/fabianguidobaldi/sezzle-challenge/backend/internal/calculator"
)

const shutdownTimeout = 10 * time.Second

func main() {
        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()
        os.Exit(run(ctx, os.Args[1:], os.Getenv, os.Stdout, os.Stderr))
}

// run is main without the process globals, so it can be tested. It returns
// the process exit code.
func run(ctx context.Context, args []string, getenv func(string) string, stdout, stderr io.Writer) int {
        flags := flag.NewFlagSet("server", flag.ContinueOnError)
        flags.SetOutput(stderr)
        healthcheck := flags.Bool("healthcheck", false, "probe /healthz on the local server and exit")
        if err := flags.Parse(args); err != nil {
                return 2
        }

        port := getenv("PORT")
        if port == "" {
                port = "8080"
        }

        if *healthcheck {
                if err := checkHealth("http://localhost:" + port + "/healthz"); err != nil {
                        fmt.Fprintln(stderr, err)
                        return 1
                }
                return 0
        }

        logger := slog.New(slog.NewJSONHandler(stdout, nil))
        ln, err := net.Listen("tcp", ":"+port)
        if err != nil {
                logger.Error("listen failed", "err", err)
                return 1
        }
        logger.Info("server listening", "addr", ln.Addr().String())
        if err := serve(ctx, logger, ln); err != nil {
                logger.Error("server failed", "err", err)
                return 1
        }
        logger.Info("server stopped")
        return 0
}

// serve runs the API on ln until ctx is cancelled, then shuts down gracefully,
// giving in-flight requests up to shutdownTimeout to finish.
func serve(ctx context.Context, logger *slog.Logger, ln net.Listener) error {
        srv := &http.Server{
                Handler:           api.NewRouter(calculator.New(), logger),
                ReadHeaderTimeout: 5 * time.Second,
                ReadTimeout:       10 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
                ErrorLog:          slog.NewLogLogger(logger.Handler(), slog.LevelError),
        }

        errCh := make(chan error, 1)
        go func() { errCh <- srv.Serve(ln) }()

        select {
        case err := <-errCh:
                return err
        case <-ctx.Done():
        }

        shutdownCtx, cancel := context.WithTimeout(context.Background(), shutdownTimeout)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil {
                return fmt.Errorf("shutdown: %w", err)
        }
        if err := <-errCh; !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        return nil
}

// checkHealth returns an error unless url answers 200 OK.
func checkHealth(url string) error {
        client := &http.Client{Timeout: 2 * time.Second}
        resp, err := client.Get(url)
        if err != nil {
                return fmt.Errorf("healthcheck: %w", err)
        }
        defer resp.Body.Close()
        if resp.StatusCode != http.StatusOK {
                return fmt.Errorf("healthcheck: unexpected status %d", resp.StatusCode)
        }
        return nil
}

package api

import (
        "log/slog"
        "net/http"

        "github.com/fabianguidobaldi/sezzle-challenge/backend/internal/calculator"
)

// Calculator performs a single calculation. It is defined here, on the
// consumer side, so handlers can be tested with a stub.
type Calculator interface {
        Calculate(op calculator.Operation, operands []string) (calculator.Result, error)
}

// calculateResponse is the body of a successful calculate response.
type calculateResponse struct {
        Operation calculator.Operation `json:"operation"`
        Operands  []string             `json:"operands"` // echoed as received
        Result    string               `json:"result"`
        Exact     bool                 `json:"exact"`
}

// handleCalculate serves POST /api/v1/calculate.
func handleCalculate(calc Calculator, logger *slog.Logger) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                req, err := decodeCalculateRequest(w, r)
                if err != nil {
                        writeFailure(w, logger, err)
                        return
                }
                res, err := calc.Calculate(req.Operation, req.Operands)
                if err != nil {
                        writeFailure(w, logger, err)
                        return
                }
                writeJSON(w, http.StatusOK, calculateResponse{
                        Operation: req.Operation,
                        Operands:  req.Operands,
                        Result:    res.Value,
                        Exact:     res.Exact,
                })
        }
}

import (

        "net/http"
        "runtime/debug"
        "time"

// recoverPanics turns a panic in next into a logged 500 JSON response.
// http.ErrAbortHandler is re-panicked, since it deliberately aborts the
// response.
func recoverPanics(logger *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                defer func() {
                        v := recover()
                        if v == nil {
                                return
                        }
                        if v == http.ErrAbortHandler {
                                panic(v)
                        }
                        logger.Error("panic serving request",
                                slog.String("method", r.Method),
                                slog.String("path", r.URL.Path),
                                slog.Any("panic", v),
                                slog.String("stack", string(debug.Stack())),
                        )
                        writeError(w, http.StatusInternalServerError, codeInternalError, internalErrorMessage)
                }()
                next.ServeHTTP(w, r)
        })
}

// logRequests logs the method, path, status and duration of each request.
func logRequests(logger *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
                next.ServeHTTP(rec, r)
                logger.LogAttrs(r.Context(), slog.LevelInfo, "request",
                        slog.String("method", r.Method),
                        slog.String("path", r.URL.Path),
                        slog.Int("status", rec.status),
                        slog.Duration("duration", time.Since(start)),
                )
        })
}

// statusRecorder captures the status code written through it. It starts at
// 200, the status net/http sends when a handler never calls WriteHeader.
type statusRecorder struct {
        http.ResponseWriter
        status      int
        wroteHeader bool
}

func (r *statusRecorder) WriteHeader(status int) {
        if !r.wroteHeader {
                r.status = status
                r.wroteHeader = true
        }
        r.ResponseWriter.WriteHeader(status)
}

func (r *statusRecorder) Write(b []byte) (int, error) {
        r.wroteHeader = true
        return r.ResponseWriter.Write(b)
}

import (

// maxBodyBytes bounds the size of a calculate request body.
const maxBodyBytes = 4 << 10

// errInvalidOperand is returned by operand.UnmarshalJSON for any JSON value
// that is neither a string nor a number.
var errInvalidOperand = errors.New("each operand must be a decimal string or a number")

// operand is a calculator operand as sent by the client. It accepts a JSON
// string or a JSON number; numbers keep their original text, so they are never
// converted to a float and lose no precision.
type operand string

// UnmarshalJSON implements json.Unmarshaler. It rejects null, booleans,
// objects and arrays.
func (o *operand) UnmarshalJSON(data []byte) error {
        // The decoder only passes complete, valid JSON values, so the first byte
        // identifies the value's type.
        switch c := data[0]; {
        case c == '"':
                return json.Unmarshal(data, (*string)(o))
        case c == '-' || (c >= '0' && c <= '9'):
                *o = operand(data)
        default:
                return errInvalidOperand
        }
        return nil
}

// calculateRequest is a decoded and validated calculate request.
type calculateRequest struct {
        Operation calculator.Operation
        Operands  []string
}

// requestError is a request that was rejected before reaching the calculator.
// It carries everything needed to write the error response.
type requestError struct {
        status  int
        code    string
        message string
}

func (e *requestError) Error() string { return e.message }

// invalidRequest returns a 400 INVALID_REQUEST error with the given message.
func invalidRequest(format string, args ...any) *requestError {
        return &requestError{status: http.StatusBadRequest, code: codeInvalidRequest, message: fmt.Sprintf(format, args...)}
}

// decodeCalculateRequest reads and validates the body of a calculate request.
// Every error it returns is a *requestError.
func decodeCalculateRequest(w http.ResponseWriter, r *http.Request) (calculateRequest, error) {
        mediaType, _, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
        if err != nil || mediaType != "application/json" {
                return calculateRequest{}, &requestError{
                        status:  http.StatusUnsupportedMediaType,
                        code:    codeUnsupportedMediaType,
                        message: "Content-Type must be application/json",
                }
        }

        // Pointers tell a missing or null field apart from a zero value.
        var body struct {
                Operation *string   `json:"operation"`
                Operands  []operand `json:"operands"`
        }
        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxBodyBytes))
        dec.DisallowUnknownFields()
        if err := dec.Decode(&body); err != nil {
                return calculateRequest{}, decodeError(err)
        }
        if err := dec.Decode(&struct{}{}); !errors.Is(err, io.EOF) {
                var maxBytesErr *http.MaxBytesError
                if errors.As(err, &maxBytesErr) {
                        return calculateRequest{}, decodeError(err)
                }
                return calculateRequest{}, invalidRequest("request body must contain a single JSON object")
        }

        switch {
        case body.Operation == nil:
                return calculateRequest{}, invalidRequest("operation is required")
        case *body.Operation == "":
                return calculateRequest{}, invalidRequest("operation must not be empty")
        case body.Operands == nil: // missing or null; [] is left to the calculator
                return calculateRequest{}, invalidRequest("operands is required and must be an array")
        }

        operands := make([]string, len(body.Operands))
        for i, o := range body.Operands {
                operands[i] = string(o)
        }
        return calculateRequest{Operation: calculator.Operation(*body.Operation), Operands: operands}, nil
}

// decodeError translates a json.Decoder error into a client-facing error.
func decodeError(err error) *requestError {
        var (
                maxBytesErr *http.MaxBytesError
                syntaxErr   *json.SyntaxError
                typeErr     *json.UnmarshalTypeError
        )
        switch {
        case errors.As(err, &maxBytesErr):
                return &requestError{
                        status:  http.StatusRequestEntityTooLarge,
                        code:    codeRequestTooLarge,
                        message: fmt.Sprintf("request body must be at most %d bytes", maxBodyBytes),
                }
        case errors.Is(err, io.EOF):
                return invalidRequest("request body must not be empty")
        case errors.As(err, &syntaxErr), errors.Is(err, io.ErrUnexpectedEOF):
                return invalidRequest("request body is not valid JSON")
        case errors.As(err, &typeErr):
                if typeErr.Field == "" {
                        return invalidRequest("request body must be a JSON object")
                }
                return invalidRequest("%s has the wrong type: got a JSON %s", typeErr.Field, typeErr.Value)
        case errors.Is(err, errInvalidOperand):
                return invalidRequest("%s", errInvalidOperand)
        case strings.HasPrefix(err.Error(), "json: unknown field "):
                // encoding/json has no typed error for unknown fields.
                return invalidRequest("%s", strings.TrimPrefix(err.Error(), "json: "))
        default:
                return invalidRequest("invalid request body")
        }
}

import (

// Error codes returned in the "code" field of error responses.
const (
        codeInvalidRequest       = "INVALID_REQUEST"
        codeUnknownOperation     = "UNKNOWN_OPERATION"
        codeInvalidOperandCount  = "INVALID_OPERAND_COUNT"
        codeInvalidNumber        = "INVALID_NUMBER"
        codeNotFound             = "NOT_FOUND"
        codeMethodNotAllowed     = "METHOD_NOT_ALLOWED"
        codeRequestTooLarge      = "REQUEST_TOO_LARGE"
        codeUnsupportedMediaType = "UNSUPPORTED_MEDIA_TYPE"
        codeDivisionByZero       = "DIVISION_BY_ZERO"
        codeUndefinedResult      = "UNDEFINED_RESULT"
        codeResultOutOfRange     = "RESULT_OUT_OF_RANGE"
        codeInternalError        = "INTERNAL_ERROR"
)

// internalErrorMessage is the only message clients see for unexpected errors;
// the details are logged instead.
const internalErrorMessage = "internal server error"

// domainErrors maps the calculator's errors to HTTP responses. The 400s mean
// the request is malformed; the 422s mean it is well formed but the math has
// no answer.
var domainErrors = []struct {
        err    error
        status int
        code   string
}{
        {calculator.ErrUnknownOperation, http.StatusBadRequest, codeUnknownOperation},
        {calculator.ErrOperandCount, http.StatusBadRequest, codeInvalidOperandCount},
        {calculator.ErrInvalidNumber, http.StatusBadRequest, codeInvalidNumber},
        {calculator.ErrDivisionByZero, http.StatusUnprocessableEntity, codeDivisionByZero},
        {calculator.ErrUndefined, http.StatusUnprocessableEntity, codeUndefinedResult},
        {calculator.ErrOutOfRange, http.StatusUnprocessableEntity, codeResultOutOfRange},
}

// errorResponse is the body of every error response.
type errorResponse struct {
        Error errorDetail `json:"error"`
}

type errorDetail struct {
        Code    string `json:"code"`
        Message string `json:"message"`
}

// writeJSON writes v as a JSON response with the given status.
func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        // Response types always encode; an error here means the client went away.
        _ = json.NewEncoder(w).Encode(v)
}

// writeError writes an error response with the given status, code and message.
func writeError(w http.ResponseWriter, status int, code, message string) {
        writeJSON(w, status, errorResponse{Error: errorDetail{Code: code, Message: message}})
}

// writeFailure writes the response for an error from request decoding or from
// the calculator. Errors it doesn't recognize are logged and reported as a
// generic 500, so internal details never reach the client.
func writeFailure(w http.ResponseWriter, logger *slog.Logger, err error) {
        var reqErr *requestError
        if errors.As(err, &reqErr) {
                writeError(w, reqErr.status, reqErr.code, reqErr.message)
                return
        }
        for _, m := range domainErrors {
                if errors.Is(err, m.err) {
                        writeError(w, m.status, m.code, err.Error())
                        return
                }
        }
        logger.Error("unexpected error", slog.Any("error", err))
        writeError(w, http.StatusInternalServerError, codeInternalError, internalErrorMessage)
}

// NewRouter returns the HTTP handler for the whole API. Every response it
// writes, including 404 and 405, is JSON.
func NewRouter(calc Calculator, logger *slog.Logger) http.Handler {
        mux := http.NewServeMux()
        // Routes are registered without a method so a wrong method gets a JSON
        // 405; the standard mux would reply in plain text.
        mux.Handle("/api/v1/calculate", allowMethod(http.MethodPost, handleCalculate(calc, logger)))
        mux.Handle("/healthz", allowMethod(http.MethodGet, handleHealth))
        mux.HandleFunc("/", handleNotFound)

        return recoverPanics(logger, logRequests(logger, mux))
}

// allowMethod rejects requests whose method isn't method with a JSON 405.
func allowMethod(method string, next http.HandlerFunc) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                if r.Method != method {
                        w.Header().Set("Allow", method)
                        writeError(w, http.StatusMethodNotAllowed, codeMethodNotAllowed,
                                "method "+r.Method+" is not allowed; use "+method)
                        return
                }
                next(w, r)
        }
}

// handleHealth serves GET /healthz.
func handleHealth(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, http.StatusOK, map[string]string{"status": "ok"})
}

// handleNotFound serves every path without a route.
func handleNotFound(w http.ResponseWriter, r *http.Request) {
        writeError(w, http.StatusNotFound, codeNotFound, "no route for "+r.URL.Path)
}

// Package calculator implements exact decimal arithmetic for the calculator API.
//
// The package is pure domain logic: it has no knowledge of HTTP or JSON. Numbers
// cross its boundary as decimal strings, so callers never depend on the
// underlying decimal library.
package calculator

import (
        "errors"
        "fmt"

        "github.com/cockroachdb/apd/v3"
)

// Operation identifies a calculator operation.
type Operation string

// Supported operations.
const (
        Add       Operation = "add"
        Subtract  Operation = "subtract"
        Multiply  Operation = "multiply"
        Divide    Operation = "divide"
        Power     Operation = "power"
        Sqrt      Operation = "sqrt"
        Percent   Operation = "percent"    // x / 100
        PercentOf Operation = "percent_of" // base × rate / 100
)

// Result is the outcome of a successful calculation.
type Result struct {
        Value string // canonical decimal string; see formatResult
        Exact bool   // false if the value may have been rounded
}

// Number format shared by operands and results: plain decimals (no exponent
// notation) with at most Precision significant digits, at most Precision
// integer digits (so every number is below 10^16 in magnitude) and at most
// MaxDecimalPlaces decimal places. Results that don't fit are rounded
// half-even, or rejected as out of range if they are too large.
const (
        Precision        = 16
        MaxDecimalPlaces = 16
)

// workingPrecision is the precision used while computing. It holds any sum or
// product of two operands exactly (at most 2×16 digits plus a carry), so those
// results are rounded only once, to the number format.
const workingPrecision = 2*Precision + 2

// Exponent range while computing. Operands in the number format keep every
// operation but power well inside it; power handles leaving it itself.
const (
        minExponent = -9999
        maxExponent = 9999
)

// newContext returns the arithmetic context shared by every operation.
func newContext() apd.Context {
        return apd.Context{
                Precision:   workingPrecision,
                Rounding:    apd.RoundHalfEven,
                MinExponent: minExponent,
                MaxExponent: maxExponent,
                Traps:       apd.Overflow | apd.Underflow | apd.DivisionByZero | apd.DivisionUndefined | apd.InvalidOperation,
        }
}

// Calculator evaluates operations on decimal strings. It holds no mutable
// state after construction, so a single Calculator is safe for concurrent
// use: every call allocates its own decimals and only reads the context.
type Calculator struct {
        ctx apd.Context
}

// New returns a Calculator using the shared arithmetic context.
func New() *Calculator {
        return &Calculator{ctx: newContext()}
}

// operation describes how many operands an operation takes and how to
// compute it. apply returns the apd condition so Calculate can derive both
// the domain error and the Exact flag.
type operation struct {
        arity int
        apply func(c *Calculator, args []*apd.Decimal) (*apd.Decimal, apd.Condition, error)
}

// registry maps every supported operation to its implementation.
// Adding an operation means adding one function and one entry here.
var registry = map[Operation]operation{
        Add:       {2, add},
        Subtract:  {2, subtract},
        Multiply:  {2, multiply},
        Divide:    {2, divide},
        Power:     {2, power},
        Sqrt:      {1, sqrt},
        Percent:   {1, percent},
        PercentOf: {2, percentOf},
}

// Calculate validates the operation and operands, computes the result and
// formats it. Errors wrap one of the sentinels in errors.go, or are an
// *OperandError for a rejected operand.
func (c *Calculator) Calculate(op Operation, operands []string) (Result, error) {
        o, ok := registry[op]
        if !ok {
                return Result{}, fmt.Errorf("%w: %q", ErrUnknownOperation, op)
        }
        if len(operands) != o.arity {
                return Result{}, fmt.Errorf("%w: %s expects %d %s, got %d",
                        ErrOperandCount, op, o.arity, plural(o.arity, "operand"), len(operands))
        }

        args := make([]*apd.Decimal, len(operands))
        for i, s := range operands {
                d, err := parseOperand(i, s)
                if err != nil {
                        return Result{}, err
                }
                args[i] = d
        }

        d, cond, err := o.apply(c, args)
        if err := domainError(op, d, cond, err); err != nil {
                return Result{}, err
        }
        roundCond, err := roundResult(&c.ctx, d)
        if err != nil {
                return Result{}, err
        }
        return Result{Value: formatResult(d), Exact: !(cond | roundCond).Inexact()}, nil
}

// domainError maps the outcome of an apd computation to a domain error, or
// nil if d is a usable result. Errors already in domain terms (from checks
// done before calling apd) pass through unchanged.
func domainError(op Operation, d *apd.Decimal, cond apd.Condition, err error) error {
        switch {
        case cond.DivisionByZero() || cond.DivisionUndefined():
                return ErrDivisionByZero
        case cond.InvalidOperation():
                return ErrUndefined
        case err != nil:
                if isDomainError(err) {
                        return err
                }
                return fmt.Errorf("calculator: %s: %w", op, err)
        case d.Form != apd.Finite:
                // Safety net: apd can return Infinity or NaN without raising a condition.
                return ErrUndefined
        }
        return nil
}

// isDomainError reports whether err already wraps one of the package's sentinels.
func isDomainError(err error) bool {
        for _, target := range []error{ErrDivisionByZero, ErrUndefined, ErrOutOfRange} {
                if errors.Is(err, target) {
                        return true
                }
        }
        return false
}

// plural returns word with an "s" appended unless n is 1.
func plural(n int, word string) string {
        if n == 1 {
                return word
        }
        return word + "s"
}

package calculator

import (
        "fmt"
        "strings"

        "github.com/cockroachdb/apd/v3"
)

// maxOperandLength bounds the raw input size before any parsing work is done.
const maxOperandLength = 64

// parseOperand converts the operand at the given index into a decimal.
// Operands must already be in the number format (see Precision); they are
// rejected, never rounded. Every error is an *OperandError.
func parseOperand(index int, s string) (*apd.Decimal, error) {
        reject := func(reason string) error {
                return &OperandError{Index: index, Value: s, Reason: reason}
        }

        switch {
        case s == "":
                return nil, reject("must not be empty")
        case len(s) > maxOperandLength:
                return nil, reject(fmt.Sprintf("must be at most %d characters", maxOperandLength))
        case strings.ContainsAny(s, "eE"):
                return nil, reject("must be a plain decimal number, without an exponent")
        }

        d, _, err := apd.NewFromString(s)
        if err != nil {
                return nil, reject("is not a valid decimal number")
        }
        if d.Form != apd.Finite {
                return nil, reject("must be a finite number")
        }

        // Trailing zeros are not significant: "1.000" has one digit.
        d.Reduce(d)
        switch {
        case d.NumDigits() > Precision:
                return nil, reject(fmt.Sprintf("must have at most %d significant digits", Precision))
        case d.Exponent < -MaxDecimalPlaces:
                return nil, reject(fmt.Sprintf("must have at most %d decimal places", MaxDecimalPlaces))
        case adjustedExponent(d) >= Precision:
                return nil, reject(fmt.Sprintf("must have at most %d integer digits", Precision))
        }
        return d, nil
}

// roundResult rounds d in place to the number format: at most Precision
// significant digits and MaxDecimalPlaces decimal places, so values too small
// for the format become 0. It returns an ErrOutOfRange error if the rounded
// value has more than Precision integer digits.
func roundResult(ctx *apd.Context, d *apd.Decimal) (apd.Condition, error) {
        if d.IsZero() {
                return 0, nil
        }
        var cond apd.Condition
        exp := max(adjustedExponent(d)-(Precision-1), -MaxDecimalPlaces)
        if int64(d.Exponent) < exp {
                var err error
                if cond, err = ctx.Quantize(d, d, int32(exp)); err != nil {
                        return cond, err
                }
        }
        // Rounding can carry into a new digit (9999999999999999.5 → 1E+16).
        if !d.IsZero() && adjustedExponent(d) >= Precision {
                return cond, errTooLarge
        }
        return cond, nil
}

// formatResult renders a value in the number format as a canonical plain
// decimal string: trailing zeros removed and no negative zero.
func formatResult(d *apd.Decimal) string {
        if d.IsZero() {
                return "0"
        }
        var r apd.Decimal
        r.Reduce(d)
        return r.Text('f')
}

// adjustedExponent is the exponent of d's most significant digit
// (e.g. 3 for 1234, -2 for 0.0123).
func adjustedExponent(d *apd.Decimal) int64 {
        return int64(d.Exponent) + d.NumDigits() - 1
}

package calculator

import (
        "errors"
        "fmt"
)

// Errors returned by Calculate. Callers match them with errors.Is; the
// returned errors wrap these with a more specific message.
var (
        ErrUnknownOperation = errors.New("unknown operation")
        ErrOperandCount     = errors.New("wrong number of operands")
        ErrInvalidNumber    = errors.New("invalid number")
        ErrDivisionByZero   = errors.New("division by zero")
        ErrUndefined        = errors.New("result is undefined")
        ErrOutOfRange       = errors.New("result out of range")
)

// errTooLarge is ErrOutOfRange for results that don't fit in Precision
// integer digits.
var errTooLarge = fmt.Errorf("%w: results must be less than 10^%d in magnitude", ErrOutOfRange, Precision)

// OperandError reports an operand that is not an acceptable decimal number.
// It wraps ErrInvalidNumber.
type OperandError struct {
        Index  int    // zero-based position in the operands list
        Value  string // the rejected input
        Reason string
}

func (e *OperandError) Error() string {
        return fmt.Sprintf("operands[%d]: %s", e.Index, e.Reason)
}

func (e *OperandError) Unwrap() error { return ErrInvalidNumber }

package calculator

import "github.com/cockroachdb/apd/v3"

// apd computes x^y with a fractional y through exp and ln, which always
// flags the result as inexact, even for 4^0.5 = 2. exactPower recovers the
// cases that are in fact exact by checking the rounded result with exact
// integer arithmetic.

// Limits that keep the check to a few thousand digits of integer arithmetic.
// Beyond them, results simply keep apd's conservative inexact flag.
const (
        maxExactDenominator = 100  // y = num/den with den ≤ 100: covers .5, .25, .2, .125, ...
        maxExactDigits      = 2000 // digits of x^|num|
)

// exactCtx multiplies without rounding for any value within the limits above:
// r^den has at most Precision × maxExactDenominator digits, x^|num| at most
// maxExactDigits, and their product at most the sum.
var exactCtx = apd.Context{
        Precision:   Precision*maxExactDenominator + maxExactDigits,
        MaxExponent: apd.MaxExponent,
        MinExponent: apd.MinExponent,
}

// exactPower reports whether x^y, which apd computed as d and flagged
// inexact, is exactly representable in the number format. If so, it returns
// that value. With y = num/den in lowest terms and r = d rounded to the
// format, x^y = r exactly if and only if r^den = x^num, which only needs
// integer powers of finite decimals and can be checked without rounding.
func exactPower(ctx *apd.Context, x, y, d *apd.Decimal) (*apd.Decimal, bool) {
        num, den, ok := smallFraction(y)
        if !ok || x.NumDigits()*abs(num) > maxExactDigits {
                return nil, false
        }

        r := new(apd.Decimal).Set(d)
        if _, err := roundResult(ctx, r); err != nil || r.IsZero() {
                return nil, false
        }

        var lhs, rhs apd.Decimal
        if !exactly(exactCtx.Pow(&lhs, r, apd.New(den, 0))) || !exactly(exactCtx.Pow(&rhs, x, apd.New(abs(num), 0))) {
                return nil, false
        }
        if num < 0 {
                // r^den = 1/x^|num|  ⇔  r^den × x^|num| = 1
                if !exactly(exactCtx.Mul(&lhs, &lhs, &rhs)) {
                        return nil, false
                }
                return r, lhs.Cmp(one) == 0
        }
        return r, lhs.Cmp(&rhs) == 0
}

// smallFraction writes a non-integer y as num/den in lowest terms. It fails
// for integers (apd already reports their powers' exactness correctly) and
// for denominators above maxExactDenominator.
func smallFraction(y *apd.Decimal) (num, den int64, ok bool) {
        var r apd.Decimal
        r.Reduce(y)
        if r.Exponent >= 0 {
                return 0, 0, false
        }
        // Operands have at most 16 significant digits and 16 decimal places, so
        // both the coefficient and 10^places fit in an int64.
        coeff := r
        coeff.Exponent = 0
        num, err := coeff.Int64()
        if err != nil {
                return 0, 0, false
        }
        den = 1
        for range -r.Exponent {
                den *= 10
        }
        g := gcd(abs(num), den)
        num, den = num/g, den/g
        return num, den, den <= maxExactDenominator
}

// exactly reports whether an exactCtx operation succeeded without rounding.
func exactly(cond apd.Condition, err error) bool {
        return err == nil && !cond.Inexact()
}

func gcd(a, b int64) int64 {
        for b != 0 {
                a, b = b, a%b
        }
        return a
}

func abs(n int64) int64 {
        if n < 0 {
                return -n
        }
        return n
}

package calculator

import (
        "fmt"

        "github.com/cockroachdb/apd/v3"
)

// Constants used by the operations. They are never modified.
var (
        one     = apd.New(1, 0)
        hundred = apd.New(100, 0)
)

// binary adapts a two-operand apd.Context method to an operation.
func binary(f func(ctx *apd.Context, d, x, y *apd.Decimal) (apd.Condition, error)) func(*Calculator, []*apd.Decimal) (*apd.Decimal, apd.Condition, error) {
        return func(c *Calculator, args []*apd.Decimal) (*apd.Decimal, apd.Condition, error) {
                d := new(apd.Decimal)
                cond, err := f(&c.ctx, d, args[0], args[1])
                return d, cond, err
        }
}

var (
        add      = binary((*apd.Context).Add)
        subtract = binary((*apd.Context).Sub)
        multiply = binary((*apd.Context).Mul)
        divide   = binary((*apd.Context).Quo)
)

// power computes x^y. A zero base with a negative exponent is rejected before
// calling apd, which answers it with Infinity and no condition. Large
// exponents need no limit: apd's cost grows with log(y).
//
// Results beyond the working exponent range come back as overflow or
// underflow, but apd doesn't always pick the right one (0.5^-1E+15 is
// reported as underflow), so the math decides: if the true result tends to 0
// it rounds to 0 like any other tiny result; otherwise it is out of range.
//
// For a fractional y, apd flags every result inexact; exactPower recognises
// the ones that are exact after all (4^0.5 = 2).
func power(c *Calculator, args []*apd.Decimal) (*apd.Decimal, apd.Condition, error) {
        x, y := args[0], args[1]
        if x.IsZero() && y.Negative && !y.IsZero() {
                return nil, 0, fmt.Errorf("%w: zero raised to a negative power", ErrDivisionByZero)
        }
        d := new(apd.Decimal)
        cond, err := c.ctx.Pow(d, x, y)
        if cond.Overflow() || cond.Underflow() {
                if tendsToZero(x, y) {
                        return new(apd.Decimal), apd.Inexact | apd.Rounded, nil
                }
                return nil, 0, errTooLarge
        }
        if err == nil && cond.Inexact() {
                if r, ok := exactPower(&c.ctx, x, y, d); ok {
                        return r, 0, nil
                }
        }
        return d, cond, err
}

// tendsToZero reports whether |x^y| < 1, i.e. |x| < 1 with y > 0 or |x| > 1
// with y < 0.
func tendsToZero(x, y *apd.Decimal) bool {
        var absX apd.Decimal
        absX.Abs(x)
        cmp := absX.Cmp(one)
        return (cmp < 0 && y.Sign() > 0) || (cmp > 0 && y.Sign() < 0)
}

// sqrt computes √x. Negative inputs are rejected up front with a clearer
// message than apd's invalid operation.
func sqrt(c *Calculator, args []*apd.Decimal) (*apd.Decimal, apd.Condition, error) {
        x := args[0]
        if x.Negative && !x.IsZero() {
                return nil, 0, fmt.Errorf("%w: square root of a negative number", ErrUndefined)
        }
        d := new(apd.Decimal)
        cond, err := c.ctx.Sqrt(d, x)
        return d, cond, err
}

// percent computes x / 100.
func percent(c *Calculator, args []*apd.Decimal) (*apd.Decimal, apd.Condition, error) {
        d := new(apd.Decimal)
        cond, err := c.ctx.Quo(d, args[0], hundred)
        return d, cond, err
}

// percentOf computes base × rate / 100 ("rate% of base"). Dividing the rate
// first is exact (it only shifts the decimal point), so the product is the
// only step that can round. The conditions of both steps are combined.
func percentOf(c *Calculator, args []*apd.Decimal) (*apd.Decimal, apd.Condition, error) {
        rate, cond, err := percent(c, args[1:])
        if err != nil {
                return rate, cond, err
        }
        d := new(apd.Decimal)
        mulCond, err := c.ctx.Mul(d, args[0], rate)
        return d, cond | mulCond, err
}
